function doGet(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); // 1. Obtener catálogos y banners (tu lógica actual) var catalogos = obtenerCatalogos(ss); var banners = obtenerBanners(ss); // 2. Buscar automáticamente los PDFs en la carpeta de Drive var folderId = "13-zebU3qkvf_LGFRpmepEoe9ANOhD6iT"; // ID de tu carpeta var folder = DriveApp.getFolderById(folderId); var files = folder.getFiles(); var pdfs = { general: "", bajaj: "", harkes: "", bujias: "" }; while (files.hasNext()) { var file = files.next(); var name = file.getName().toUpperCase(); var url = file.getDownloadUrl(); // O file.getUrl() if (name.includes("GENERAL") && !pdfs.general) pdfs.general = url; if (name.includes("BAJAJ") && !pdfs.bajaj) pdfs.bajaj = url; if (name.includes("HARKES") && !pdfs.harkes) pdfs.harkes = url; if (name.includes("BALINERAS-BUJIAS") && !pdfs.bujias) pdfs.bujias = url; } var respuesta = { catalogos: catalogos, banners: banners, pdfs: pdfs }; return ContentService.createTextOutput(JSON.stringify(respuesta)) .setMimeType(ContentService.MimeType.JSON); }
